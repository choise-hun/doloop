## 3. 새 파일 — `backend/src/main/kotlin/com/ulevelp/api/lib/LocaleResolver.kt` (전체 코드)

```kotlin
package com.ulevelp.api.lib

/**
 * L1-1: 다국어 코드('ko', 'en', 'ja') 검증 및 Accept-Language 헤더 파서.
 * 근거: 계획/상용_서비스_계획.md §3-4, 개발산출물/명세/L1-1_구현명세.md §1 D3~D4
 */
object LocaleResolver {
    val SUPPORTED_LOCALES: Set<String> = setOf("ko", "en", "ja")

    fun isValidLocale(value: String): Boolean {
        return value in SUPPORTED_LOCALES
    }

    private data class LangCandidate(
        val code: String,
        val q: Double,
        val order: Int,
    )

    /**
     * RFC 9110 Accept-Language 헤더(예: "ko-KR,ko;q=0.9,en-US;q=0.8,en;q=0.7")를 파싱하여
     * 지원 언어('ko', 'en', 'ja') 중 가중치(q)가 가장 높은 첫 번째 언어 코드를 반환한다.
     * 매칭되는 지원 언어가 없거나 헤더가 비어 있으면 null을 반환한다.
     */
    fun parseAcceptLanguage(header: String?): String? {
        if (header.isNullOrBlank()) return null
        val candidates = mutableListOf<LangCandidate>()

        header.split(',').forEachIndexed { idx, rawToken ->
            val token = rawToken.trim()
            if (token.isEmpty()) return@forEachIndexed

            val parts = token.split(';').map { it.trim() }.filter { it.isNotEmpty() }
            if (parts.isEmpty()) return@forEachIndexed

            val range = parts[0]
            if (range == "*") return@forEachIndexed

            var q = 1.0
            for (i in 1 until parts.size) {
                val param = parts[i]
                if (param.startsWith("q=", ignoreCase = true)) {
                    val parsed = param.substring(2).trim().toDoubleOrNull() ?: return@forEachIndexed
                    if (parsed <= 0.0 || parsed > 1.0) return@forEachIndexed
                    q = parsed
                }
            }

            val primary = range
                .substringBefore('-')
                .substringBefore('_')
                .trim()
                .lowercase()

            if (primary in SUPPORTED_LOCALES) {
                candidates.add(LangCandidate(code = primary, q = q, order = idx))
            }
        }

        return candidates
            .sortedWith(compareByDescending<LangCandidate> { it.q }.thenBy { it.order })
            .firstOrNull()
            ?.code
    }
}
```

class LocaleApiTest {

    private val ecKey = ECKeyGenerator(Curve.P_256).keyID("locale-test-kid").generate()
    private val jwkSet = JWKSet(ecKey.toPublicJWK())
    private val issuer = "https://example.supabase.co/auth/v1"
    private val json = Json { ignoreUnknownKeys = true }

    private val jwtConfig = JwtVerifierConfig(
        issuer = issuer,
        audience = "authenticated",
        jwkProvider = { jwkSet }
    )

    private fun signToken(userId: UUID, email: String = "locale@example.com"): String {
        val now = Instant.now()
        val claims = JWTClaimsSet.Builder()
            .issuer(issuer)
            .audience("authenticated")
            .subject(userId.toString())
            .claim("email", email)
            .claim("role", "authenticated")
            .issueTime(Date.from(now))
            .expirationTime(Date.from(now.plusSeconds(3600)))
            .build()
        val jwt = SignedJWT(
            JWSHeader.Builder(JWSAlgorithm.ES256).keyID(ecKey.keyID).build(),
            claims
        )
        jwt.sign(ECDSASigner(ecKey))
        return jwt.serialize()
    }

    private fun createIsolatedDb() = DatabaseFactory.initInMemory(
        "locale_test_${UUID.randomUUID().toString().replace("-", "")}"
    )

    @Test
    fun `L1-1-1 LocaleResolver parseAcceptLanguage parses weights and supported codes`() {
        assertEquals("ko", LocaleResolver.parseAcceptLanguage("ko-KR,ko;q=0.9,en-US;q=0.8,en;q=0.7"))
        assertEquals("ja", LocaleResolver.parseAcceptLanguage("ja,en-US;q=0.9,ko;q=0.8"))
        assertEquals("en", LocaleResolver.parseAcceptLanguage("fr-CH, fr;q=0.9, en-GB;q=0.8, ja;q=0.7, *;q=0.5"))
        assertEquals("ja", LocaleResolver.parseAcceptLanguage("en;q=0, ja-JP;q=0.5"))
        assertNull(LocaleResolver.parseAcceptLanguage("es-ES,fr;q=0.8,de;q=0.7"))
        assertNull(LocaleResolver.parseAcceptLanguage("*;q=0.8"))
        assertNull(LocaleResolver.parseAcceptLanguage(""))
        assertNull(LocaleResolver.parseAcceptLanguage("   "))
        assertNull(LocaleResolver.parseAcceptLanguage(null))

        assertTrue(LocaleResolver.isValidLocale("ko"))
        assertTrue(LocaleResolver.isValidLocale("en"))
        assertTrue(LocaleResolver.isValidLocale("ja"))
        assertTrue(!LocaleResolver.isValidLocale("fr"))
        assertTrue(!LocaleResolver.isValidLocale("KO"))
    }

    @Test
    fun `L1-1-2 first signup sets locale from Accept-Language without overwriting existing profile`() = testApplication {
        val db = createIsolatedDb()
        val profileService = ProfileService(db)
        application {
            module(jwtVerifierConfig = jwtConfig, profileService = profileService)
        }

        // 1) 일본어 브라우저로 첫 진입한 사용자 -> locale == "ja"
        val userJa = UUID.randomUUID()
        val tokenJa = signToken(userJa, "ja-user@example.com")
        val firstRes = client.get("/api/v1/me") {
            header(HttpHeaders.Authorization, "Bearer $tokenJa")
            header(HttpHeaders.AcceptLanguage, "ja-JP,ja;q=0.9,en-US;q=0.8")
        }
        assertEquals(HttpStatusCode.OK, firstRes.status)
        val firstJson = json.parseToJsonElement(firstRes.bodyAsText()).jsonObject
        assertEquals("ja", firstJson["locale"]?.jsonPrimitive?.content)

        // 2) 같은 사용자가 다른 Accept-Language(en-US)로 다시 요청해도 기존 "ja"가 유지됨
        val secondRes = client.get("/api/v1/me") {
            header(HttpHeaders.Authorization, "Bearer $tokenJa")
            header(HttpHeaders.AcceptLanguage, "en-US,en;q=0.9")
        }
        assertEquals(HttpStatusCode.OK, secondRes.status)
        val secondJson = json.parseToJsonElement(secondRes.bodyAsText()).jsonObject
        assertEquals("ja", secondJson["locale"]?.jsonPrimitive?.content)

        // 3) 미지원 언어(fr-FR)로 첫 진입한 사용자 -> locale == null (자동 모드)
        val userFr = UUID.randomUUID()
        val tokenFr = signToken(userFr, "fr-user@example.com")
        val frRes = client.get("/api/v1/me") {
            header(HttpHeaders.Authorization, "Bearer $tokenFr")
            header(HttpHeaders.AcceptLanguage, "fr-FR,fr;q=0.9")
        }
        assertEquals(HttpStatusCode.OK, frRes.status)
        val frJson = json.parseToJsonElement(frRes.bodyAsText()).jsonObject
        assertTrue(frJson["locale"] == null || frJson["locale"] is JsonNull)
    }

    @Test
    fun `L1-1-3 PATCH me updates locale to ko en ja and resets to null`() = testApplication {
        val db = createIsolatedDb()
        val profileService = ProfileService(db)
        application {
            module(jwtVerifierConfig = jwtConfig, profileService = profileService)
        }

        val userId = UUID.randomUUID()
        val token = signToken(userId, "patch-locale@example.com")

        for (lang in listOf("en", "ja", "ko")) {
            val patchRes = client.patch("/api/v1/me") {
                header(HttpHeaders.Authorization, "Bearer $token")
                contentType(ContentType.Application.Json)
                setBody("""{"locale":"$lang"}""")
            }
            assertEquals(HttpStatusCode.OK, patchRes.status)
            val patchBody = json.parseToJsonElement(patchRes.bodyAsText()).jsonObject
            assertEquals(lang, patchBody["locale"]?.jsonPrimitive?.content)

            val getRes = client.get("/api/v1/me") {
                header(HttpHeaders.Authorization, "Bearer $token")
            }
            assertEquals(HttpStatusCode.OK, getRes.status)
            val getBody = json.parseToJsonElement(getRes.bodyAsText()).jsonObject
            assertEquals(lang, getBody["locale"]?.jsonPrimitive?.content)
        }

        // null 전달 시 자동 모드(null)로 복귀
        val resetRes = client.patch("/api/v1/me") {
            header(HttpHeaders.Authorization, "Bearer $token")
            contentType(ContentType.Application.Json)
            setBody("""{"locale":null}""")
        }
        assertEquals(HttpStatusCode.OK, resetRes.status)
        val resetBody = json.parseToJsonElement(resetRes.bodyAsText()).jsonObject
        assertTrue(resetBody["locale"] == null || resetBody["locale"] is JsonNull)
    }

    @Test
    fun `L1-1-4 PATCH me rejects unsupported or malformed locale with 400 INVALID_INPUT`() = testApplication {
        val db = createIsolatedDb()
        val profileService = ProfileService(db)
        application {
            module(jwtVerifierConfig = jwtConfig, profileService = profileService)
        }

        val userId = UUID.randomUUID()
        val token = signToken(userId, "bad-locale@example.com")

        val invalidBodies = listOf(
            """{"locale":"fr"}""",
            """{"locale":""}""",
            """{"locale":"   "}""",
            """{"locale":123}""",
            """{"locale":true}""",
            """{}"""
        )

        for (raw in invalidBodies) {
            val res = client.patch("/api/v1/me") {
                header(HttpHeaders.Authorization, "Bearer $token")
                contentType(ContentType.Application.Json)
                setBody(raw)
            }
            assertEquals(HttpStatusCode.BadRequest, res.status, "Expected 400 for body: $raw")
            val errObj = json.parseToJsonElement(res.bodyAsText()).jsonObject["error"]!!.jsonObject
            assertEquals("INVALID_INPUT", errObj["code"]?.jsonPrimitive?.content)
        }
    }

- 새 Flyway 마이그레이션 파일(`V6__*.sql`, `V9__*.sql`) 생성 금지 (`V5__tiers_credits_and_community.sql:14`에 이미 반영됨)

- `GeminiSchemas`에서 `locale == "ko"`일 때 기존 프롬프트 텍스트 변경 금지 (`NoteSvgAndSchemaTest`의 `sv5`·`sv5b`가 깨짐)
- Gemini 프롬프트 언어 지시에서 `difficulty`(`입문`/`중급`/`고급`), `instrument`(`건반`/`기타`), `source`(`말함`/`화면`/`추정`) enum 값을 영어나 일본어로 번역하라고 지시하는 행위 금지 (DB CHECK 제약 위반 발생)
