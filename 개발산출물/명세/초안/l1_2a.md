   export function checkChord(raw: string): ChordChip {
     const symbol = raw.trim();
     if (symbol.length > MAX_SYMBOL) {
       return { symbol, ok: false, warning: t('prog.warn.maxLen', { max: MAX_SYMBOL }) };
     }
     const r = parseChordSymbol(symbol);
     if (!r.ok) return { symbol, ok: false, warning: t(WARN_KEY[r.problem]) };
     if (r.warnings && r.warnings.length > 0) {
       const pairs = r.warnings.map((w) => `${w.redundant} = ${w.kept}`).join(', ');
       return { symbol, ok: true, warning: t('prog.warn.dupDegree', { pairs }) };
     }
     return { symbol, ok: true, warning: '' };

function stripCommentsAndIgnored(src) {
