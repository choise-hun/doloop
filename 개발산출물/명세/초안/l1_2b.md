## 3. 다국어 엔진 및 사전 코드 (`i18n.mts`, `ko.mts` 추가분, `en.mts`, `ja.mts`)

### 3-1. `frontend/src/lib/i18n.mts` (전체 코드)

```ts
// L0 + L1-2b: 다국어 엔진 — t(key, vars), 자동 언어 감지(detectAutoLang), 수동/자동 설정(setLang), 프로필 동기화(syncLangFromProfile)
// 규칙: 순수 .mts 모듈(React import 없음 — node --test에서 직접 import 가능)
import { KO } from './i18n/ko.mts';
import { EN } from './i18n/en.mts';
import { JA } from './i18n/ja.mts';

export type LocaleCode = 'ko' | 'en' | 'ja';
export type LocalePref = 'auto' | LocaleCode;
export type I18nKey = keyof typeof KO;
export type I18nVars = Record<string, string | number>;

export const LANG_STORAGE_KEY = 'ulp:lang';
export const LANG_EVENT = 'ulp:langchange';

const DICTS: Record<LocaleCode, Record<I18nKey, string>> = {
  ko: KO,
  en: EN,
  ja: JA,
};

const SUPPORTED: ReadonlySet<string> = new Set<LocaleCode>(['ko', 'en', 'ja']);

export function isLocaleCode(v: unknown): v is LocaleCode {
  return typeof v === 'string' && SUPPORTED.has(v);
}

function readStoredPref(): LocalePref {
  try {
    if (typeof localStorage === 'undefined') return 'auto';
    const raw = localStorage.getItem(LANG_STORAGE_KEY);
    return isLocaleCode(raw) ? raw : 'auto';
  } catch {
    return 'auto';
  }
}

function writeStoredPref(pref: LocalePref): void {
  try {
    if (typeof localStorage === 'undefined') return;
    if (pref === 'auto') {
      localStorage.removeItem(LANG_STORAGE_KEY);
    } else if (isLocaleCode(pref)) {
      localStorage.setItem(LANG_STORAGE_KEY, pref);
    }
  } catch {
    /* 저장 불가 환경 무시 */
  }
}

export interface AutoLangEnv {
  languages?: ReadonlyArray<string>;
  language?: string;
  timeZone?: string;
}

/**
 * 브라우저 언어 목록(navigator.languages) -> 타임존(Asia/Seoul, Asia/Tokyo) -> 'ko' 순으로 자동 언어를 판정한다.
 */
export function detectAutoLang(env?: AutoLangEnv): LocaleCode {
  const langs: string[] = [];
  if (env && Array.isArray(env.languages) && env.languages.length > 0) {
    langs.push(...env.languages);
  } else if (env && typeof env.language === 'string' && env.language) {
    langs.push(env.language);
  } else if (typeof navigator !== 'undefined') {
    if (Array.isArray(navigator.languages) && navigator.languages.length > 0) {
      langs.push(...navigator.languages);
    } else if (typeof navigator.language === 'string' && navigator.language) {
      langs.push(navigator.language);
    }
  }

  for (const tag of langs) {
    const primary = String(tag || '')
      .trim()
      .split(/[-_]/)[0]
      .toLowerCase();
    if (isLocaleCode(primary)) {
      return primary;
    }
  }

  let tz = env?.timeZone;
  if (tz === undefined) {
    try {
      if (typeof Intl !== 'undefined' && typeof Intl.DateTimeFormat === 'function') {
        tz = Intl.DateTimeFormat().resolvedOptions().timeZone;
      }
    } catch {
      tz = '';
    }
  }
  if (tz === 'Asia/Tokyo') return 'ja';
  if (tz === 'Asia/Seoul') return 'ko';

  return 'ko';
}

let profileLocaleCache: LocaleCode | null = null;

function resolveActiveLang(): LocaleCode {
  const pref = readStoredPref();
  if (pref !== 'auto') return pref;
  if (profileLocaleCache) return profileLocaleCache;
  if (typeof window === 'undefined') return 'ko';
  return detectAutoLang();
}

let current: LocaleCode = resolveActiveLang();
const listeners = new Set<(lang: LocaleCode) => void>();

function syncHtmlLang(lang: LocaleCode): void {
  try {
    if (typeof document !== 'undefined' && document.documentElement) {
      document.documentElement.lang = lang;
    }
  } catch {
    /* DOM 미존재 환경 무시 */
  }
}

function notifyListeners(next: LocaleCode): void {
  syncHtmlLang(next);
  listeners.forEach((fn) => {
    try {
      fn(next);
    } catch {
      /* 리스너 오류 격리 */
    }
  });
  try {
    if (typeof window !== 'undefined' && typeof window.dispatchEvent === 'function' && typeof Event === 'function') {
      window.dispatchEvent(new Event(LANG_EVENT));
    }
  } catch {
    /* 이벤트 디스패치 불가 환경 무시 */
  }
}

syncHtmlLang(current);

export function getLang(): LocaleCode {
  return current;
}

export function getLangPref(): LocalePref {
  return readStoredPref();
}

/**
 * 사용자의 언어 선택('auto' | 'ko' | 'en' | 'ja')을 즉시 반영하고 활성 언어 코드를 반환한다.
 */
export function setLang(pref: LocalePref, env?: AutoLangEnv): LocaleCode {
  writeStoredPref(pref);
  if (pref === 'auto') {
    current = profileLocaleCache || detectAutoLang(env);
  } else if (isLocaleCode(pref)) {
    current = pref;
  }
  notifyListeners(current);
  return current;
}

/**
 * 서버 /api/v1/me 응답의 locale 값을 반영한다.
 * 기기에 수동 설정(ulp:lang = 'ko'|'en'|'ja')이 있으면 수동 설정이 우선한다.
 */
export function syncLangFromProfile(profileLocale: string | null | undefined): LocaleCode {
  profileLocaleCache = isLocaleCode(profileLocale) ? profileLocale : null;
  const pref = readStoredPref();
  if (pref === 'auto') {
    const next = profileLocaleCache || (typeof window !== 'undefined' ? detectAutoLang() : 'ko');
    if (next !== current) {
      current = next;
      notifyListeners(current);
    } else {
      syncHtmlLang(current);
    }
  }
  return current;
}

export function onLangChange(fn: (lang: LocaleCode) => void): () => void {
  if (typeof fn !== 'function') return () => {};
  listeners.add(fn);
  return () => {
    listeners.delete(fn);
  };
}

export function t(key: I18nKey, vars?: I18nVars): string {
  const dict = DICTS[current] || KO;
  const s: unknown = dict[key] ?? KO[key];
  if (typeof s !== 'string') return String(key);
  if (!vars) return s;
  return s.replace(/\{(\w+)\}/g, (m, name: string) =>
    Object.prototype.hasOwnProperty.call(vars, name) ? String(vars[name]) : m
  );
}
```
