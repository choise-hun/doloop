## 3. 시그니처 (`frontend/src/lib/noteMarkdown.mts`)

```ts
export interface NoteMarkdownInput {          // NoteDetailDto에서 쓰는 필드만 (apiClient.ts를 import하지 않는다)
  title?: string; url?: string; videoId?: string; channel?: string; createdAt?: string;
  tags?: string[]; memo?: string; headline?: string; points?: string[]; keywords?: string[];
  analysis?: Record<string, any>;
}
/** practice.html?v=<id>[&t=<초>] — 상대 경로. origin을 주면 절대 URL */
export function notePracticeHref(videoId: string, sec?: number | null, origin?: string): string;
/** 노트 → 마크다운 문자열(끝 줄바꿈 1개). origin = 링크 절대화용(페이지는 window.location.origin) */
export function buildNoteMarkdown(note: NoteMarkdownInput, origin: string): string;
```

- 초 해석: node 테스트가 `.tsx`를 import할 수 없으므로 `noteMarkdown.mts` 안에 비공개 함수 `tsSec`를 둔다 — `NoteDetailPage.tsx:12-23` 본문을 그대로 복사(주석에 출처 한 줄). `NoteDetailPage`의 `parseTimestampSec`는 그대로 둔다(`r1_settings_notes.test.js`가 소스에서 뽑음)
- `NoteDetailPage.tsx`는 `TimestampCell`·주요 구간·액션 바에서 `notePracticeHref(videoId, sec)`를 쓴다(origin 없이 = 지금과 같은 상대 경로)

## 6. 비동기 골격 — 마크다운 복사 (이 흐름만)

```ts
// 상태: copyBusy, copyMsg {text, kind}; ref: copySeqRef(0), copyTimerRef(number|null)
const clearCopyTimer = () => { if (copyTimerRef.current) { clearTimeout(copyTimerRef.current); copyTimerRef.current = null; } };

const copyMarkdown = async () => {
  if (!note || copyBusy) return;
  const mySeq = ++copySeqRef.current;          // 이 클릭의 번호
  const myUid = activeUid;
  const myNoteId = note.id;
  const text = buildNoteMarkdown(note, window.location.origin);   // await 전에 스냅샷
  clearCopyTimer();
  setCopyBusy(true);
  let ok = false;
  try {
    if (!navigator.clipboard?.writeText) throw new Error('no clipboard');
    await navigator.clipboard.writeText(text);
    ok = true;
  } catch {
    ok = false;                                 // 문구로 알린다(아래)
  } finally {
    // 늦은 결과: 다른 클릭·노트 재로딩·계정 전환이 끼었으면 아무것도 세우지 않는다
    const stale = copySeqRef.current !== mySeq
      || auth.getUser()?.id !== myUid
      || noteIdRef.current !== myNoteId;        // noteIdRef = setNote 때 같이 갱신
    if (!stale) {
      setCopyBusy(false);
      setCopyMsg(ok ? { text: t('noteDetail.copyMdDone'), kind: 'ok' } : { text: t('noteDetail.copyMdFail'), kind: 'err' });
      if (ok) copyTimerRef.current = window.setTimeout(() => setCopyMsg({ text: '', kind: '' }), 2500);
    }
  }
};
// clearState(:109)에 추가: ++copySeqRef.current; clearCopyTimer(); setCopyBusy(false); setCopyMsg(빈값); noteIdRef.current = null
// 언마운트: useEffect(() => () => clearCopyTimer(), [])
```

- `stale`일 때 `setCopyBusy(false)`도 하지 않는 이유: 끼어든 쪽(`clearState`)이 이미 풀었다. 다른 클릭이 끼어드는 일은 `copyBusy`로 막혀 사실상 재로딩·계정 전환뿐
- 수집 토글(`toggleCollect`)은 흐름을 바꾸지 않는다. 위치만 옮긴다
