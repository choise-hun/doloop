```ts
// M4-1: 코드 재해석 탭 프리셋 — 표준 화성학 이름(특정 강의 순서·이름을 쓰지 않는다)
// frets 순서 = 6번줄(저음 E) → 1번줄, -1 = 안 침. 해석 결과는 test/m41_reint_presets.test.js가 엔진으로 확인한다
export interface ReintPreset {
  id: string;
  label: string;
  /** 칩 title(툴팁) — 이 손 모양의 기준 해석 한 줄 */
  hint: string;
  frets: number[];
}

export const REINT_PRESETS: ReadonlyArray<ReintPreset> = [
  { id: 'drop2', label: 'Drop 2 · maj7', hint: 'Gmaj7 — 1·3·5·7에서 위에서 둘째 음을 한 옥타브 내림', frets: [-1, -1, 5, 4, 3, 2] },
  { id: 'quartal', label: 'Quartal · 4도 쌓기', hint: 'A–D–G–C 완전4도 쌓기 — D7sus4·Am7(11)', frets: [-1, -1, 7, 7, 8, 8] },
  { id: 'rootless9', label: 'Rootless 9 · 3-7-9-5', hint: 'Eb7(9) 근음 없는 보이싱 — 아래부터 3·b7·9·5', frets: [-1, -1, 5, 6, 6, 6] },
  { id: 'rootless13', label: 'Rootless 13 · 7-3-13-9', hint: 'B7(9,13) 근음 없는 보이싱 — 아래부터 b7·3·13·9', frets: [-1, -1, 7, 8, 9, 9] },
  { id: 'm6', label: 'm6 = m7♭5', hint: 'Gm6과 Em7(b5)가 같은 손 모양', frets: [-1, -1, 2, 3, 3, 3] },
];

export const DEFAULT_REINT_PRESET_ID = 'drop2';
```

| rootless9 | G Db F Bb | Db6(#11), Gm7(b5), Bbm6, C7sus4(b9), Eb7(9), A7(b9,b13) | 6 |

## 4. 테스트 — `test/m41_reint_presets.test.js` (전체 코드)

```js
// M4-1: 코드 재해석 프리셋 — 표준 이름·엔진 해석 확인 (명세: 개발산출물/명세/M4-1_구현명세.md)
'use strict';
const { test, before } = require('node:test');
const assert = require('node:assert/strict');

let P;
let CT;
before(async () => {
  P = await import('../frontend/src/lib/chords/reintPresets.mts');
  CT = (await import('../frontend/src/lib/engine/theory.mjs')).default;
});

function names(frets) {
  const notes = frets
    .map((f, s) => (f < 0 ? null : { s, f, pitch: CT.OPEN[s] + f }))
    .filter(Boolean);
  return CT.interpret(notes).map((i) => i.name);
}

test('M41-1 프리셋 5개 · id·label 중복 없음 · 6줄 · 기본 id 존재', () => {
  assert.equal(P.REINT_PRESETS.length, 5);
  assert.equal(new Set(P.REINT_PRESETS.map((p) => p.id)).size, 5);
  assert.equal(new Set(P.REINT_PRESETS.map((p) => p.label)).size, 5);
  for (const p of P.REINT_PRESETS) {
    assert.equal(p.frets.length, 6, p.id);
    assert.ok(p.hint.length > 0, p.id);
  }
  assert.ok(P.REINT_PRESETS.some((p) => p.id === P.DEFAULT_REINT_PRESET_ID));
});

test('M41-2 각 프리셋의 기준 해석이 엔진 결과에 있다', () => {
  const expect = {
    drop2: ['Gmaj7'],
    quartal: ['D7sus4', 'Am7(11)'],
    rootless9: ['Eb7(9)'],
    rootless13: ['B7(9,13)'],
    m6: ['Gm6', 'Em7(b5)'],
  };
  for (const p of P.REINT_PRESETS) {
    const got = names(p.frets);
    for (const n of expect[p.id]) assert.ok(got.includes(n), `${p.id}: ${n} 없음 — ${got.join(', ')}`);
  }
});

test('M41-3 옛 이름(모양 n)이 남지 않음', () => {
  for (const p of P.REINT_PRESETS) assert.doesNotMatch(p.label, /모양\s*\d/);
});
```

- 잘못된 입력 확인(명세 체크리스트에 있음): M41-2 기대값 하나를 일부러 `'Gmaj9'`로 바꿔 실패 문구를 본 뒤 되돌린다. 실패 출력 한 줄을 결과 파일에 붙인다
